**Goal:** Produce offline SHAP explanations of the deployed DDQN DDA policy on the 98 real beta-test decisions behind BCM 15.31%, outputting waterfall/beeswarm/counterfactual plots + a written failure-pattern report for the thesis.

**Architecture:** Parse 98 `dda_event` records from the real closed-beta logs (`DataPost/*.jsonl`) into `states.npy[98,6]` + `actions.npy` + per-decision `survival.npy`/`outcomes.npy`. """Single source of truth for SHAP XAI pipeline. See spec §2, §4.1."""

OBS_SIZE = 6

    "HP Ratio",            # dda_obs_snapshot[0]  (== player_hp_ratio, cross-checked)
    "Turn Count",          # [1] turn/15
    "Player Level",        # [2] level/5
    "Dmg Dealt Ratio",     # [3] areaTotalEnemyHP / damageDealt
    "QTE Accuracy",        # [4] successfulQTE / totalQTE
    "Resource Depletion",  # [5]
]

ACTION_NAME_TO_INT = {
    "Very Easy": 0, "Easy": 1, "Normal": 2, "Hard": 3, "Very Hard": 4,
}
ACTION_INT_TO_NAME = {v: k for k, v in ACTION_NAME_TO_INT.items()}

# outcome_label from survival_ratio = hp_final / hp_initial (spec §4.1)
OUTCOME_CODES = {"Subjugate": 0, "Balanced": 1, "Rebellious": 2}
OUTCOME_INT_TO_NAME = {v: k for k, v in OUTCOME_CODES.items()}
SR_BALANCED = (0.4, 0.6)  # inclusive


def survival_to_outcome(sr: float) -> int:
    """Return outcome int code from survival_ratio. spec §4.1."""
    lo, hi = SR_BALANCED
    if sr < lo:
        return OUTCOME_CODES["Rebellious"]
    if sr > hi:
        return OUTCOME_CODES["Subjugate"]
    return OUTCOME_CODES["Balanced"]

# tools/xai/parse_dda_logs.py
"""Parse real closed-beta dda_event logs into numpy arrays. Spec §2, §4.1, §5.

dda_event ↔ battle alignment (spec §12 open item, resolved by ts order):
  events are sorted by timestamp within a session; a dda_event (decision for the
  upcoming battle) is paired with the NEXT battle_start (hp_initial) and the next
  battle_end (hp_final) after it. survival_ratio = hp_final / hp_initial.
"""
import json, os, glob
from collections import defaultdict
import numpy as np
from tools.xai import constants as C


def _load_events(paths):
    """Read jsonl paths, return list of (ts, session_id, event_type, payload)."""
    evs = []
    for p in paths:
        with open(p, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line or line.startswith("#"):
                    continue
                try:
                    d = json.loads(line)
                except json.JSONDecodeError:
                    continue
                evs.append((d.get("ts", ""), d.get("session_id", ""),
                           d.get("event_type", ""), d.get("payload", {}) or {}))
    return evs


def _player_hp(payload):
    pp = payload.get("player_performance", {}) or {}
    return pp.get("player_hp")


def parse_dda_logs(log_paths, out_dir=None):
    """Parse dda_event + aligned battles -> arrays dict. Spec §5.

    log_paths: list of .jsonl file paths (e.g. DataPost/*.jsonl).
    out_dir: if given, write states/actions/survival/outcomes.npy + meta.json.
    Returns dict with numpy arrays.
    """
    evs = _load_events(log_paths)
    # group by session, sort by ts
    by_session = defaultdict(list)
    for ts, sid, et, pl in evs:
        by_session[sid].append((ts, et, pl))
    for sid in by_session:
        by_session[sid].sort(key=lambda t: t[0])

    states, actions, survival, outcomes, hp_ratios = [], [], [], [], []
    meta_decisions = []  # traceability

    for sid, sess in by_session.items():
        pending = None  # dict for the most recent dda_event awaiting its battle
        hp_initial = None
        for ts, et, pl in sess:
            if et == "dda_event":
                if pending is not None:
                    # previous decision never got a battle_end; skip it (no outcome)
                    pass
                snap = pl.get("dda_obs_snapshot")
                name = pl.get("dda_action_taken")
                if snap is None or len(snap) != C.OBS_SIZE or name not in C.ACTION_NAME_TO_INT:
                    continue
                pending = {
                    "obs": list(snap),
                    "action": C.ACTION_NAME_TO_INT[name],
                    "hp_ratio": pl.get("player_hp_ratio"),
                    "ts": ts,
                }
                hp_initial = None
            elif et == "battle_start":
                hp_initial = _player_hp(pl)
            elif et == "battle_end" and pending is not None:
                hp_final = _player_hp(pl)
                if hp_initial is None or hp_initial == 0 or hp_final is None:
                    pending = None
                    continue
                sr = float(hp_final) / float(hp_initial)
                sr = min(1.0, max(0.0, sr))
                states.append(pending["obs"])
                actions.append(pending["action"])
                survival.append(sr)
                outcomes.append(C.survival_to_outcome(sr))
                hp_ratios.append(pending["hp_ratio"])
                meta_decisions.append({
                    "session_id": sid, "decision_ts": pending["ts"],
                    "action": C.ACTION_INT_TO_NAME[pending["action"]],
                    "hp_initial": hp_initial, "hp_final": hp_final,
                    "survival_ratio": sr,
                    "outcome": C.OUTCOME_INT_TO_NAME[C.survival_to_outcome(sr)],
                })
                pending = None

    states = np.asarray(states, dtype=np.float32)
    actions = np.asarray(actions, dtype=np.int64)
    survival = np.asarray(survival, dtype=np.float32)
    outcomes = np.asarray(outcomes, dtype=np.int64)
    hp_ratios = np.asarray([hp if hp is not None else np.nan for hp in hp_ratios],
                           dtype=np.float32)

    # range validation (spec §8): clamp obs to [0,1], warn on violation
    n_viol = int(np.sum((states < 0) | (states > 1)))
    if n_viol:
        print(f"[parse_dda_logs] WARN {n_viol} obs values out of [0,1]; clamping")
        states = np.clip(states, 0.0, 1.0)

    out = {
        "states": states, "actions": actions, "survival": survival,
        "outcomes": outcomes, "hp_ratios": hp_ratios,
    }
    if out_dir:
        os.makedirs(out_dir, exist_ok=True)
        np.save(os.path.join(out_dir, "states.npy"), states)
        np.save(os.path.join(out_dir, "actions.npy"), actions)
        np.save(os.path.join(out_dir, "survival.npy"), survival)
        np.save(os.path.join(out_dir, "outcomes.npy"), outcomes)
        meta = {
            "n_decisions": int(states.shape[0]),
            "feature_names": C.FEATURE_NAMES,
            "decisions": meta_decisions,
        }
        with open(os.path.join(out_dir, "meta.json"), "w", encoding="utf-8") as f:
            json.dump(meta, f, indent=2)
        with open(os.path.join(out_dir, "log_paths.txt"), "w", encoding="utf-8") as f:
            f.write("\n".join(str(p) for p in log_paths))
    return out


def main():
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--log-dir", default=r"E:\COLLEGE\SKOM\Implementasi\Battle Logs\DataPost")
    ap.add_argument("--out-dir", default="tools/xai")
    a = ap.parse_args()
    paths = sorted(glob.glob(os.path.join(a.log_dir, "*.jsonl")))
    if not paths:
        raise SystemExit(f"no jsonl in {a.log_dir}")
    out = parse_dda_logs(paths, out_dir=a.out_dir)
    print(f"parsed {out['states'].shape[0]} decisions from {len(paths)} files")
    # spec §9.7 traceability: action distribution
    for k, v in C.ACTION_INT_TO_NAME.items():
        n = int(np.sum(out["actions"] == k))
        print(f"  {v}: {n}")


if __name__ == "__main__":
    main()
```

# tools/xai/shap_net.py
"""Minimal PyTorch replica of QNetworkDDQN + ONNX weight extraction. Spec §6.

Architecture: Norm(running mean/var) -> Linear(6,128) -> ReLU -> Linear(128,128)
-> ReLU -> Linear(128,5). Weights extracted from the deployed beta .onnx.
"""
import numpy as np
import onnx
import onnxruntime as ort
import torch
import torch.nn as nn


class ShapQNet(nn.Module):
    def __init__(self, norm_mean, norm_var, eps, w1, b1, w2, b2, w3, b3):
        super().__init__()
        self.register_buffer("norm_mean", torch.as_tensor(norm_mean, dtype=torch.float32))
        self.register_buffer("norm_var", torch.as_tensor(norm_var, dtype=torch.float32))
        self.eps = float(eps)
        self.l1 = nn.Linear(w1.shape[0], w1.shape[1])
        self.l2 = nn.Linear(w2.shape[0], w2.shape[1])
        self.l3 = nn.Linear(w3.shape[0], w3.shape[1])
        # Linear weight in PyTorch is [out, in]; onnx Gemm weight is often [out,in]
        # or [in,out] -- load_from_onnx transposes as needed. Here we trust caller.
        with torch.no_grad():
            self.l1.weight.copy_(torch.as_tensor(w1, dtype=torch.float32))
            self.l1.bias.copy_(torch.as_tensor(b1, dtype=torch.float32))
            self.l2.weight.copy_(torch.as_tensor(w2, dtype=torch.float32))
            self.l2.bias.copy_(torch.as_tensor(b2, dtype=torch.float32))
            self.l3.weight.copy_(torch.as_tensor(w3, dtype=torch.float32))
            self.l3.bias.copy_(torch.as_tensor(b3, dtype=torch.float32))

    def forward(self, x):
        x = (x - self.norm_mean) / torch.sqrt(self.norm_var + self.eps)
        x = torch.relu(self.l1(x))
        x = torch.relu(self.l2(x))
        return self.l3(x)  # [B,5]


def _initializers(onnx_path):
    m = onnx.load(onnx_path)
    inits = {init.name: numpy_from_onnx(init) for init in m.graph.initializer}
    return inits, m


def numpy_from_onnx(init):
    arr = np.frombuffer(init.raw_data, dtype=_np_dtype(init.data_type)).copy()
    arr = arr.reshape(init.dims)
    return arr.astype(np.float32)


def _np_dtype(onnx_dtype):
    from onnx import TensorProto as T
    return {
        T.FLOAT: np.float32, T.DOUBLE: np.float64,
        T.INT32: np.int32, T.INT64: np.int64,
    }.get(onnx_dtype, np.float32)


def _find_weight_pair(inits, candidate_names_w, candidate_names_b, in_shape, out_shape):
    """Find a (weight, bias) pair by name patterns or by shape [out,in]/[in,out]."""
    for nw in candidate_names_w:
        if nw in inits and inits[nw].ndim == 2:
            w = inits[nw]
            if w.shape == (out_shape, in_shape):
                return w, inits.get(candidate_names_b[0], np.zeros(out_shape, np.float32))
            if w.shape == (in_shape, out_shape):
                return w.T, inits.get(candidate_names_b[0], np.zeros(out_shape, np.float32))
    # fallback: any 2D initializer with matching total elements
    want = in_shape * out_shape
    for name, arr in inits.items():
        if arr.ndim == 2 and arr.size == want:
            w = arr if arr.shape == (out_shape, in_shape) else arr.T
            return w, np.zeros(out_shape, np.float32)
    raise KeyError(f"no weight pair {in_shape}->{out_shape} found")


def load_from_onnx(onnx_path):
    """Extract weights from a beta .onnx into ShapQNet. Spec §6.

    Raises KeyError if mapping fails. Caller should fall back to KernelExplainer.
    """
    inits, m = _initializers(onnx_path)

    # eps constant: search graph for a scalar constant near normalization
    eps = 1e-5  # ML-Agents default; spec §12 open item (confirm from onnx)
    for n in m.graph.node:
        if n.op_type == "Constant":
            for attr in n.attribute:
                if attr.name == "value" and attr.t.data_type in (1, 11):
                    arr = numpy_from_onnx(attr.t)
                    if arr.size == 1 and 1e-8 < float(arr.flat[0]) < 1e-2:
                        eps = float(arr.flat[0])

    # normalization running mean/var: shape [6] tensors.
    mean = var = None
    for name, arr in inits.items():
        if arr.shape == (6,):
            if "running_var" in name or ("var" in name and var is None):
                var = arr
            elif "running_mean" in name or ("mean" in name and mean is None):
                mean = arr
    if mean is None or var is None:
        # pick by order among [6]-shape tensors
        sixes = [(n, a) for n, a in inits.items() if a.shape == (6,)]
        if len(sixes) >= 2:
            mean, var = sixes[0][1], sixes[1][1]
    if mean is None or var is None:
        raise KeyError("normalization running mean/var not found in onnx initializers")

    # linears: 6->128, 128->128, 128->5. Match by shape among 2D initializers.
    two_d = [(n, a) for n, a in inits.items() if a.ndim == 2]
    w1 = w2 = w3 = None
    for n, a in two_d:
        s = a.shape
        if s == (128, 6) or s == (6, 128):
            w1 = a if s == (128, 6) else a.T
        elif s == (128, 128):
            w2 = a
        elif s == (5, 128) or s == (128, 5):
            w3 = a if s == (5, 128) else a.T
    if w1 is None or w2 is None or w3 is None:
        raise KeyError(f"linears not found; 2D shapes seen: {[a.shape for _, a in two_d]}")

    def _bias(out_shape):
        for n, a in inits.items():
            if a.shape == (out_shape,):
                return a
        return np.zeros(out_shape, np.float32)

    net = ShapQNet(mean, var, eps,
                   w1=w1, b1=_bias(128), w2=w2, b2=_bias(128), w3=w3, b3=_bias(5))
    return net


def onnx_inference(onnx_path, states_np):
    """Run the deployed .onnx for Q-values [N,5]. Spec §9.1.

    The ML-Agents export has multiple outputs; the deterministic-discrete-action
    head is NOT the raw Q-values. We need the value-head output. To stay robust,
    this returns the output whose shape is [N,5] (the Q-value head).
    """
    sess = ort.InferenceSession(onnx_path, providers=["CPUExecutionProvider"])
    in_name = sess.get_inputs()[0].name
    outs = sess.run(None, {in_name: states_np.astype(np.float32)})
    for o in outs:
        if o is not None and o.ndim == 2 and o.shape[1] == 5:
            return o.astype(np.float32)
    raise KeyError(f"no [N,5] output in onnx; shapes: {[o.shape if o is not None else None for o in outs]}")


def check_faithfulness(shap_net, onnx_path, states_np, tol=1e-4):
    q_torch = shap_net(torch.as_tensor(states_np, dtype=torch.float32)).detach().numpy()
    q_onnx = onnx_inference(onnx_path, states_np)
    q_torch = q_torch.reshape(q_onnx.shape)
    diff = float(np.max(np.abs(q_torch - q_onnx)))
    return {"max_diff": diff, "passed": diff < tol,
            "q_torch_shape": q_torch.shape, "q_onnx_shape": q_onnx.shape}

# tools/xai/identify_model.py
"""Identify the deployed beta model by argmax match rate. Spec §9.7.

For each candidate .onnx, infer Q on the 98 beta states and compare argmax to the
logged dda_action_taken. Highest match wins. Allows ~5% epsilon-greedy random
actions (exploration_final_eps=0.05), so >=90% = the true beta model.
"""
import json, os
import numpy as np
from tools.xai import shap_net as S


def identify_model(candidate_paths, states_np, actions_np, infer_fn=None):
    """Rank candidate .onnx by argmax match rate vs logged actions.

    infer_fn(path, states) -> [N,5] Q-values; default onnx_inference (DI for tests).
    Returns {best_path, best_match_rate, rankings: [(path, rate)], verdict}.
    """
    if infer_fn is None:
        infer_fn = S.onnx_inference
    rankings = []
    for path in candidate_paths:
        try:
            q = infer_fn(path, states_np)
            pred = np.argmax(q, axis=1)
            rate = float(np.mean(pred == actions_np))
        except Exception as e:
            rate = -1.0
            print(f"[identify_model] {path}: inference failed ({e!r})")
        rankings.append((path, rate))
    rankings.sort(key=lambda r: r[1], reverse=True)
    best_path, best_rate = rankings[0]
    verdict = "pass" if best_rate >= 0.90 else ("warn" if best_rate >= 0.70 else "flag")
    return {"best_path": best_path, "best_match_rate": best_rate,
            "rankings": rankings, "verdict": verdict}


def main():
    import argparse, glob
    from tools.xai import constants as C
    ap = argparse.ArgumentParser()
    ap.add_argument("--states", default="tools/xai/states.npy")
    ap.add_argument("--actions", default="tools/xai/actions.npy")
    ap.add_argument("--models-dir", default="Assets/Resources/DDA/Models")
    ap.add_argument("--meta", default="tools/xai/meta.json")
    a = ap.parse_args()
    states = np.load(a.states)
    actions = np.load(a.actions)
    cands = sorted(glob.glob(os.path.join(a.models_dir, "*.onnx")))
    res = identify_model(cands, states, actions)
    print("=== beta-model probe (spec §9.7) ===")
    for path, rate in res["rankings"]:
        print(f"  {rate*100:5.1f}%  {os.path.basename(path)}")
    print(f"verdict: {res['verdict']}  best: {os.path.basename(res['best_path'])}")
    # merge into meta.json
    meta = json.load(open(a.meta, encoding="utf-8")) if os.path.exists(a.meta) else {}
    meta["beta_model"] = res["best_path"]
    meta["beta_match_rate"] = res["best_match_rate"]
    meta["model_rankings"] = [(os.path.basename(p), r) for p, r in res["rankings"]]
    json.dump(meta, open(a.meta, "w", encoding="utf-8"), indent=2)
    if res["verdict"] == "flag":
        raise SystemExit("HARD FLAG: no candidate matches beta policy >=70% (spec §9.7). "
                         "The deployed beta model may be missing from --models-dir.")


if __name__ == "__main__":
    main()

"""SHAP explainer CLI: GradientExplainer on ShapQNet, KernelExplainer fallback on .onnx.
Spec §3, §7, §9.1-9.4. Outputs waterfalls + beeswarm to results/shap/<model>/.
"""
import os, json, argparse
import numpy as np
import torch
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import shap

from tools.xai import constants as C
from tools.xai import shap_net as S


def build_gradient_explainer(net, background):
    """GradientExplainer (DeepSHAP). Returns shap.GradientExplainer."""
    return shap.GradientExplainer(net, torch.as_tensor(background, dtype=torch.float32))


def build_kernel_explainer(infer_fn, background):
    """KernelExplainer fallback on the raw .onnx (approach C). infer_fn: states->[N,5]."""
    return shap.KernelExplainer(infer_fn, background)


def _save_waterfall(expl, base_values, shap_vals_row, state_row, action_idx, decision_idx, out_dir):
    os.makedirs(out_dir, exist_ok=True)
    exp = shap.Explanation(
        values=np.asarray(shap_vals_row, dtype=np.float64),
        base_values=float(np.asarray(base_values).ravel()[action_idx]),
        data=state_row,
        feature_names=C.FEATURE_NAMES,
    )
    shap.waterfall_plot(exp, show=False)
    plt.tight_layout()
    p = os.path.join(out_dir, f"waterfall_decision{decision_idx}_action{action_idx}.png")
    plt.savefig(p, dpi=150)
    plt.close()
    return p


def explain_all(states, actions, expl, expected_value, out_dir, decision_indices=None,
                all_actions=True):
    """Emit waterfalls. decision_indices: which rows to explain (default all 98).
    all_actions: if True, explain all 5 action Q-values per decision; else only chosen."""
    states_t = torch.as_tensor(states, dtype=torch.float32)
    sv = expl.shap_values(states_t)  # list per output of [N,6]
    sv = [np.asarray(s) for s in sv]
    base = np.asarray(expected_value, dtype=np.float64)  # [5]
    idxs = decision_indices if decision_indices is not None else list(range(len(states)))
    for d in idxs:
        targets = range(C.ACTION_SIZE) if all_actions else [int(actions[d])]
        for a in targets:
            _save_waterfall(expl, base[a] if base.ndim == 1 else base,
                            sv[a][d] if not isinstance(sv[a], list) else sv[a][d],
                            states[d], a, d, out_dir)
    return sv, base


def explain_beeswarm(states, actions, sv, base, out_dir):
    """Beeswarm over the chosen-action SHAP. Spec §1 global layer."""
    os.makedirs(out_dir, exist_ok=True)
    chosen_sv = np.stack([sv[int(actions[i])][i] for i in range(len(states))])
    exp = shap.Explanation(
        values=chosen_sv,
        base_values=np.array([float(base[int(actions[i])]) for i in range(len(states))]),
        data=states,
        feature_names=C.FEATURE_NAMES,
    )
    shap.summary_plot(chosen_sv, states, feature_names=C.FEATURE_NAMES, show=False)
    plt.tight_layout()
    p = os.path.join(out_dir, "summary_beeswarm.png")
    plt.savefig(p, dpi=150, bbox_inches="tight")
    plt.close()
    return p, chosen_sv

    ap = argparse.ArgumentParser()
    ap.add_argument("--states", default="tools/xai/states.npy")
    ap.add_argument("--actions", default="tools/xai/actions.npy")
    ap.add_argument("--outcomes", default="tools/xai/outcomes.npy")
    ap.add_argument("--meta", default="tools/xai/meta.json")
    ap.add_argument("--out-dir", default=None)
    ap.add_argument("--decisions", default="all", help="all | representative | comma list of indices")
    ap.add_argument("--all-actions", action="store_true", default=True)
    a = ap.parse_args()

    states = np.load(a.states).astype(np.float32)
    actions = np.load(a.actions)
    meta = json.load(open(a.meta, encoding="utf-8"))
    onnx_path = meta["beta_model"]
    model_stem = os.path.splitext(os.path.basename(onnx_path))[0]
    out_dir = a.out_dir or f"results/shap/{model_stem}"
    os.makedirs(out_dir, exist_ok=True)

    # decide backend: try extraction + faithfulness; else KernelExplainer
    try:
        net = S.load_from_onnx(onnx_path)
        chk = S.check_faithfulness(net, onnx_path, states[:10])
        assert chk["passed"], chk
        expl = build_gradient_explainer(net, states)
        sv, base = explain_all(states, actions, expl, expl.expected_value, out_dir,
                                decision_indices=_decisions(a.decisions, len(states)))
        print(f"[explain_shap] GradientExplainer backend (max_diff={chk['max_diff']:.2e})")
    except Exception as e:
        print(f"[explain_shap] extraction/faithfulness failed ({e!r}); fallback KernelExplainer")
        from tools.xai.shap_net import onnx_inference
        expl = build_kernel_explainer(lambda s: onnx_inference(onnx_path, s), states)
        sv_raw = expl.shap_values(states)  # list per output
        sv = [np.asarray(s) for s in sv_raw]
        base = np.asarray(expl.expected_value, dtype=np.float64)
        explain_all(states, actions, expl, base, out_dir,
                    decision_indices=_decisions(a.decisions, len(states)), all_actions=True)
        # NOTE: explain_all expects a GradientExplainer object; for Kernel, call sv directly:

    # beeswarm (chosen-action)
    # (uses sv computed above)

    print(f"[explain_shap] wrote plots to {out_dir}")

    return [int(x) for x in spec.split(",") if x.strip() != ""]

# tools/xai/report.py
"""Generate report.md: global ranking + failure-pattern sections + counterfactual notes.
Spec §4 component 4, §4.1, §5.1.
"""
import os
import numpy as np
from tools.xai import constants as C


def _mean_abs_shap(sv):
    return np.mean(np.abs(sv), axis=0)


def _category_block(name, states, actions, outcomes, sv, code):
    mask = outcomes == code
    n = int(mask.sum())
    lines = [f"# Pola kegagalan ({name})", ""]
    if n == 0:
        lines += [f"No {name} decisions in the data (n=0) — insufficient data; section skipped (not fabricated).", ""]
        return "\n".join(lines)
    sv_c = sv[mask]
    ma = _mean_abs_shap(sv_c)
    order = np.argsort(ma)[::-1]
    lines += [f"n = {n} decisions (outcome code {code}).",
              "", "| Rank | Feature | mean |SHAP| |", "|------|---------|-------------|"]
    for r, i in enumerate(order, 1):
        lines.append(f"| {r} | {C.FEATURE_NAMES[i]} | {ma[i]:.4f} |")
    # typical action in this category
    act_c = actions[mask]
    act_name = C.ACTION_INT_TO_NAME[int(np.bincount(act_c).argmax())]
    lines += ["", f"Typical action chosen: **{act_name}**.", ""]
    return "\n".join(lines)


def generate_report(states, actions, outcomes, sv, base, model_meta, out_dir):
    os.makedirs(out_dir, exist_ok=True)
    n = states.shape[0]
    ma = _mean_abs_shap(sv)
    order = np.argsort(ma)[::-1]
    lines = [
        "# SHAP XAI Report — DDQN DDA",
        "",
        f"**Model:** `{os.path.basename(model_meta.get('beta_model','?'))}`",
        f"**Beta-model match rate:** {model_meta.get('beta_match_rate', float('nan'))*100:.1f}% (spec §9.7)",
        f"**Decisions explained:** {n}",
        "",
        "## Global feature ranking (mean |SHAP|, chosen-action)",
        "",
        "| Rank | Feature | mean |SHAP| |",
        "|------|---------|-------------|",
    ]
    for r, i in enumerate(order, 1):
        lines.append(f"| {r} | {C.FEATURE_NAMES[i]} | {ma[i]:.4f} |")
    lines += ["",
              "## Validitas (spec §5.1)",
              "",
              "SHAP explanations are computed on the real closed-beta `dda_event` states "
              "(DataPost, 98 decisions = basis BCM 15.31%). Weights are extracted from the "
              "deployed beta `.onnx` identified by the match-rate probe. Explanations are "
              "descriptive of the deployed policy on states it actually met — not causal "
              "claims about training. Counterfactuals outside the observed observation range "
              "are extrapolation.",
              ""]
    lines.append(_category_block("Subjugate", states, actions, outcomes, sv, 0))
    lines.append("")
    lines.append(_category_block("Rebellious", states, actions, outcomes, sv, 2))
    lines += ["",
              "## Counterfactual boundary notes",
              "",
              "See `counterfactual_*.png` and diff tables. Perturbations that move an "
              "observation outside the observed beta range are flagged as extrapolation.",
              ""]
    path = os.path.join(out_dir, "report.md")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines))
    return path
```

# append to tools/xai/explain_shap.py
def check_additivity(net, states, actions, tol=1e-3, nsamples=200):
    expl = build_gradient_explainer(net, states)
    sv = expl.shap_values(torch.as_tensor(states, dtype=torch.float32))
    base = np.asarray(expl.expected_value, dtype=np.float64)
    for i in range(len(states)):
        a = int(actions[i])
        recon = float(base[a]) + float(np.asarray(sv[a][i]).sum())
        q = float(net(torch.as_tensor(states[i:i+1], dtype=torch.float32))[0, a])
        if abs(recon - q) >= tol:
            return False
    return True


def check_outcome_coverage(outcomes):
    return all(int((outcomes == c).sum()) >= 1 for c in (0, 1, 2))


def run_self_check(states_path="tools/xai/states.npy",
                   actions_path="tools/xai/actions.npy",
                   outcomes_path="tools/xai/outcomes.npy",
                   survival_path="tools/xai/survival.npy",
                   meta_path="tools/xai/meta.json",
                   models_dir="Assets/Resources/DDA/Models"):
    import json
    from tools.xai import identify_model as IM
    results = {}
    states = np.load(states_path).astype(np.float32)
    actions = np.load(actions_path); outcomes = np.load(outcomes_path)
    survival = np.load(survival_path); meta = json.load(open(meta_path, encoding="utf-8"))
    onnx_path = meta.get("beta_model")

    # 9.2 range
    results["range"] = bool(states.shape[1] == 6 and np.all((states >= 0) & (states <= 1)))
    # 9.7 probe (re-run to confirm)
    probe = IM.identify_model(
        sorted(__import__("glob").glob(f"{models_dir}/*.onnx")), states, actions)
    results["beta_probe"] = probe["verdict"] in ("pass", "warn")
    results["beta_match_rate"] = probe["best_match_rate"]
    # 9.1 faithfulness
    try:
        net = S.load_from_onnx(onnx_path)
        chk = S.check_faithfulness(net, onnx_path, states[:10])
        results["faithfulness"] = bool(chk["passed"])
        results["faithfulness_max_diff"] = chk["max_diff"]
        # 9.3 additivity
        results["additivity"] = check_additivity(net, states, actions)
    except Exception as e:
        results["faithfulness"] = False
        results["faithfulness_err"] = repr(e)
        results["additivity"] = False
        net = None
    # 9.4 determinism
    if net is not None:
        q1 = net(torch.as_tensor(states[:3], dtype=torch.float32)).detach().numpy()
        q2 = net(torch.as_tensor(states[:3], dtype=torch.float32)).detach().numpy()
        results["determinism"] = bool(np.allclose(q1, q2, atol=1e-7))
    else:
        results["determinism"] = False
    # 9.5 outcome coverage
    results["outcome_coverage"] = check_outcome_coverage(outcomes)
    # 9.6 survival sanity: outcomes consistent with survival thresholds
    exp_outcomes = np.array([C.survival_to_outcome(float(s)) for s in survival], dtype=np.int64)
    results["survival_sanity"] = bool(np.array_equal(exp_outcomes, outcomes))

    print("=== self-check (spec §9) ===")
    for k, v in results.items():
        print(f"  {k}: {v}")
    all_pass = all(v in (True,) or (isinstance(v, bool) and v) for k, v in results.items()
                  if k in ("range","beta_probe","faithfulness","additivity","determinism",
                           "outcome_coverage","survival_sanity"))
    print(f"=== overall: {'PASS' if all_pass else 'FAIL'} ===")
    return all_pass, results
